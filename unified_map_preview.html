Gameplay Mechanics with Elevation:

// Get tactical advantage based on elevation public float GetTacticalAdvantage(Vector2 attackerPos, Vector2 defenderPos) { float attackerHeight = GetElevationAtPosition(attackerPos); float defenderHeight = GetElevationAtPosition(defenderPos); float heightDiff = attackerHeight - defenderHeight; if (heightDiff > 10f) return 1.5f; // 50% damage bonus (high ground) if (heightDiff > 5f) return 1.25f; // 25% damage bonus if (heightDiff < -10f) return 0.75f; // 25% penalty (attacking uphill) return 1.0f; // No modifier } // Calculate visibility range based on elevation public float GetVisibilityRange(Vector2 observerPos, float baseRange) { float elevation = GetElevationAtPosition(observerPos); float heightBonus = Mathf.Min(elevation / 10f, 2f); // Max 2x range return baseRange * (1f + heightBonus); }
